#!/usr/bin/env python3
"""R-0152 independent check of LH030 (does not use classify.py or tabulate.py).
For each read: verify each retained body's SHA-256 against its log, then per URL print status, bytes, and raw-substring
presence of CVE-2026-115560 and of 'upload_progress' in the raw bytes. Prints first read with the CVE string per URL,
Ubuntu non-404/non-200 reads, GET counts and bytes per read."""
import hashlib, json, sys
from pathlib import Path
DATA = Path(sys.argv[1])
reads = sorted(DATA.glob('read-*'))
first = {}; prev = {}
total_gets = 0; hash_bad = 0
for d in reads:
    log = json.loads((d / 'read_log.json').read_text())
    total_gets += len(log)
    recv = sum(r.get('bytes') or 0 for r in log)
    starts = [r['start_utc'] for r in log if r.get('start_utc')]; fins = [r['finish_utc'] for r in log if r.get('finish_utc')]
    print(f"{d.name}: {len(log)} GETs, {recv} bytes logged, {min(starts)[:19]} to {max(fins)[:19]}")
    for r in log:
        f = d / r['file'] if r.get('file') else None
        raw = f.read_bytes() if f and f.exists() else None
        if raw is not None and hashlib.sha256(raw).hexdigest() != r.get('sha256'):
            hash_bad += 1
            # bodies with emails redacted after hashing would differ; report
            print(f"   hash differs: {r['name']}")
        cve = raw is not None and b'CVE-2026-115560' in raw
        up = raw is not None and b'upload_progress' in raw
        key = r['name']
        state = (r.get('status'), cve)
        if key not in first and cve: first[key] = (d.name, prev.get(key))
        if r['source'] == 'ubuntu' and r.get('status') not in (200, 404):
            print(f"   ubuntu status {r.get('status')} at {r.get('start_utc')} err={r.get('error','')}")
        prev[key] = (d.name, r.get('status'))
print('total GETs', total_gets, 'hash mismatches', hash_bad)
print('first read whose raw body contains CVE-2026-115560 (and previous read, status):')
for k, v in sorted(first.items()): print(' ', k, v)
